function xorBase64(text, key) {
  const encoder = new TextEncoder();
  const bytes = encoder.encode(text);
  const keyBytes = encoder.encode(key);
  for (let i = 0; i < bytes.length; i++) bytes[i] ^= keyBytes[i % keyBytes.length];
  let binary = "";
  for (let i = 0; i < bytes.length; i++) binary += String.fromCharCode(bytes[i]);
  return btoa(binary);
}

function dexorBase64(base64, key) {
  const binary = atob(base64);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
  const encoder = new TextEncoder();
  const keyBytes = encoder.encode(key);
  for (let i = 0; i < bytes.length; i++) bytes[i] ^= keyBytes[i % keyBytes.length];
  return new TextDecoder().decode(bytes);
}

async function handleRequest(request) {
    const url = new URL(request.url);
    const reqOrigin = request.headers.get("Origin") || request.headers.get("Referer") || "";
    
    let isAllowed = true;
    let allowedOriginHeader = "*";

    if (ALLOWED_DOMAINS[0] !== "*") {
      if (reqOrigin) {
        try {
          const originUrl = new URL(reqOrigin);
          const hostname = originUrl.hostname;
          // Validasi Hostname (exact match atau subdomain)
          isAllowed = ALLOWED_DOMAINS.some(domain => hostname === domain || hostname.endsWith("." + domain));
          if (isAllowed) {
            allowedOriginHeader = originUrl.origin;
          }
        } catch(e) {
          isAllowed = false;
        }
      }
      // Blokir akses proxy dan playlist secara langsung tanpa origin (mencegah pencurian data lewat curl)
      if (!reqOrigin && (url.pathname.startsWith('/proxy') || url.pathname.startsWith('/api/playlist'))) {
         isAllowed = false;
      }
    }

    if (!isAllowed) {
      return new Response("Akses Ditolak: Domain tidak diizinkan oleh sistem keamanan web", { status: 403 });
    }

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders(allowedOriginHeader) });
    }

    if (url.pathname === "/" || url.pathname === "/index.html") {
      return new Response(HTML, {
        headers: { "content-type": "text/html; charset=UTF-8" },
      });
    }

    if (url.pathname === "/api/playlist") {
      return handlePlaylist(url, request, allowedOriginHeader);
    }

    if (url.pathname === "/proxy") {
      return handleProxy(url, request, allowedOriginHeader);
    }

    return new Response("Not found", { status: 404 });
}

async function handlePlaylist(url, request, allowedOriginHeader = "*") {
  let target = url.searchParams.get("url");
  if (!target || target === "default") {
    target = BACKEND_DEFAULT_PLAYLIST;
  }

  let resp;
  try {
    resp = await fetch(target, {
      headers: { "User-Agent": "Mozilla/5.0 (SiaranTuner)" },
    });
  } catch (e) {
    return json({ error: "Gagal mengambil playlist: " + e.message }, 502, allowedOriginHeader);
  }

  if (!resp.ok) return json({ error: `Playlist mengembalikan status ${resp.status}` }, 502, allowedOriginHeader);

  const text = await resp.text();
  if (!text.includes("#EXTM3U")) return json({ error: "File bukan playlist M3U yang valid" }, 422, allowedOriginHeader);

  const channels = parseM3U(text, target);
  if (channels.length === 0) return json({ error: "Tidak ada channel ditemukan" }, 422, allowedOriginHeader);

  const payload = JSON.stringify({ channels });
  return json({ encrypted: xorBase64(payload, SECRET_KEY) }, 200, allowedOriginHeader);
}

function parseM3U(text, baseUrl) {
  const lines = text.split(/\r?\n/);
  const channels = [];
  
  let currentItem = { drmType: "", drmKey: "", headers: {} };
  let hasExtInf = false;

  for (let raw of lines) {
    const line = raw.trim();
    if (!line) continue;

    if (line.startsWith("#EXTINF")) {
      hasExtInf = true;
      const info = line.slice(line.indexOf(":") + 1);
      const commaIdx = info.lastIndexOf(",");
      const attrsStr = commaIdx >= 0 ? info.slice(0, commaIdx) : info;
      currentItem.name = (commaIdx >= 0 ? info.slice(commaIdx + 1) : "Tanpa Nama").trim();

      const attrRegex = /([\w-]+)="([^"]*)"/g;
      let m;
      while ((m = attrRegex.exec(attrsStr))) {
        currentItem[m[1].toLowerCase()] = m[2];
      }
      
      currentItem.logo = currentItem["tvg-logo"] || currentItem["group-logo"] || "";
      currentItem.group = currentItem["group-title"] || "Lainnya";

    } else if (line.startsWith("#KODIPROP:inputstream.adaptive.license_type=")) {
      currentItem.drmType = line.substring(line.indexOf("=") + 1).trim().toLowerCase();
    } else if (line.startsWith("#KODIPROP:inputstream.adaptive.license_key=")) {
      currentItem.drmKey = line.substring(line.indexOf("=") + 1).trim();
    } else if (line.startsWith("#KODIPROP:inputstream.adaptive.stream_headers=")) {
      const hStr = line.substring(line.indexOf("=") + 1).trim();
      const hParts = hStr.split("&");
      for (const p of hParts) {
        const eqIdx = p.indexOf("=");
        if (eqIdx > 0) {
          const key = decodeURIComponent(p.substring(0, eqIdx));
          const val = decodeURIComponent(p.substring(eqIdx + 1));
          if (key.toLowerCase() === 'user-agent') currentItem.headers['User-Agent'] = val;
          else if (key.toLowerCase() === 'referer') currentItem.headers['Referer'] = val;
          else currentItem.headers[key] = val;
        }
      }
    } else if (line.startsWith("#EXTVLCOPT:http-user-agent=")) {
      currentItem.headers['User-Agent'] = line.substring(line.indexOf("=") + 1).trim();
    } else if (line.startsWith("#EXTVLCOPT:http-referrer=")) {
      currentItem.headers['Referer'] = line.substring(line.indexOf("=") + 1).trim();
    } else if (line.startsWith("#")) {
      continue; 
    } else {
      let urlPart = line;

      if (line.includes("|")) {
        const parts = line.split("|");
        urlPart = parts[0];
        for (let i = 1; i < parts.length; i++) {
          const p = parts[i];
          const eqIdx = p.indexOf("=");
          if (eqIdx > 0) {
            const key = p.substring(0, eqIdx).trim().toLowerCase();
            const val = p.substring(eqIdx + 1).trim();
            if (key === 'user-agent') currentItem.headers['User-Agent'] = val;
            if (key === 'referer') currentItem.headers['Referer'] = val;
          }
        }
      }

      try { currentItem.url = new URL(urlPart, baseUrl).toString(); } 
      catch (e) { currentItem.url = urlPart; }

      if (!currentItem.name) currentItem.name = "Tanpa Nama";
      if (!currentItem.group) currentItem.group = "Lainnya";

      channels.push({ ...currentItem });
      currentItem = { drmType: "", drmKey: "", headers: {} };
      hasExtInf = false;
    }
  }

  return channels.map((c, i) => ({ ...c, number: i + 1 }));
}

async function handleProxy(url, request, allowedOriginHeader = "*") {
  let target = "";
  let customUa = "";
  let customRef = "";
  let customHeaders = "";
  let explicitType = "";

  const q = url.searchParams.get("q");
  if (q) {
    try {
      const decodedQuery = dexorBase64(q, SECRET_KEY);
      const params = new URLSearchParams(decodedQuery);
      if (params.has("url")) target = params.get("url");
      if (params.has("ua")) customUa = params.get("ua");
      if (params.has("ref")) customRef = params.get("ref");
      if (params.has("headers")) customHeaders = params.get("headers");
      if (params.has("type")) explicitType = params.get("type");
    } catch(e) {}
  } else {
    return new Response("Akses Ditolak: URL wajib dienkripsi", { status: 403 });
  }

  if (!target) return new Response("Parameter 'url' tidak ditemukan di payload terenkripsi", { status: 400 });

  let targetOrigin = "";
  try { 
    const u = new URL(target);
    const host = u.hostname;
    
    const isIp = /^([0-9]{1,3}\.){3}[0-9]{1,3}$/.test(host);
    if (isIp) {
      const parts = host.split('.').map(Number);
      // SSRF Protection: Blokir akses ke IP Private, Loopback, dan Cloud Metadata
      if (
        parts[0] === 127 ||
        parts[0] === 10 ||
        (parts[0] === 172 && parts[1] >= 16 && parts[1] <= 31) ||
        (parts[0] === 192 && parts[1] === 168) ||
        (parts[0] === 169 && parts[1] === 254) ||
        parts[0] === 0
      ) {
        return new Response("Akses Ditolak: SSRF terdeteksi (Internal IP)", { status: 403 });
      }
      u.hostname += ".nip.io";
      target = u.toString();
    }
    targetOrigin = u.origin; 
  } 
  catch (e) { return new Response("URL tidak valid", { status: 400 }); }

  const reqHeaders = new Headers();
  
  if (customUa) {
    reqHeaders.set("User-Agent", customUa);
  } else {
    reqHeaders.set("User-Agent", request.headers.get("User-Agent") || "Mozilla/5.0 (Windows NT 10.0; Win64; x64)");
  }

  reqHeaders.set("Accept", "*/*");
  reqHeaders.set("Accept-Language", "en-US,en;q=0.9");

  const range = request.headers.get("range");
  if (range) reqHeaders.set("Range", range);

  if (customRef) {
    reqHeaders.set("Referer", customRef);
    try { 
      reqHeaders.set("Origin", new URL(customRef).origin); 
    } catch (e) {}
  }

  if (customHeaders) {
    try {
      const arr = JSON.parse(customHeaders);
      for (const h of arr) {
        const idx = h.indexOf(":");
        if (idx > 0) {
          reqHeaders.set(h.substring(0, idx).trim(), h.substring(idx + 1).trim());
        }
      }
    } catch(e) {}
  }

  let fetchOpts = { method: request.method, headers: reqHeaders, redirect: "manual" };
  let reqBodyBuf = null;
  if (request.method !== "GET" && request.method !== "HEAD") {
    reqBodyBuf = await request.arrayBuffer();
    fetchOpts.body = reqBodyBuf;
    const ct = request.headers.get("content-type") || "application/octet-stream";
    reqHeaders.set("Content-Type", ct);
  }

  // --- 🚀 BYPASS VISION+ VIA VERCEL NODE.JS PROXY ---
  const isVisionPlus = target.includes("visionplus.id") || 
                       (customRef && customRef.includes("visionplus.id")) ||
                       target.includes("cloudfront.net") || 
                       target.includes("akamaized.net");
                       
  if (isVisionPlus && typeof VISION_PROXY_URL !== "undefined" && !VISION_PROXY_URL.includes("GANTI_DENGAN_URL_VERCEL_ANDA")) {
    target = VISION_PROXY_URL + encodeURIComponent(target);
  }
  // --------------------------------------------------

  let upstream;
  let currentTarget = target;
  let redirectCount = 0;

  try {
    while (redirectCount < 5) {
      upstream = await fetch(currentTarget, fetchOpts);
      if ([301, 302, 303, 307, 308].includes(upstream.status)) {
        const loc = upstream.headers.get("location");
        if (loc) {
          currentTarget = new URL(loc, currentTarget).toString();
          if (upstream.status === 301 || upstream.status === 302 || upstream.status === 303) {
            fetchOpts.method = "GET";
            delete fetchOpts.body;
          } else {
            // 307 & 308 -> Pertahankan method & POST body!
            if (reqBodyBuf) fetchOpts.body = reqBodyBuf.slice(0); // pass fresh copy of buffer
          }
          redirectCount++;
          continue;
        }
      }
      break;
    }
  } catch (e) {
    return new Response("Gagal mengambil stream: " + e.message, { status: 502 });
  }

  const contentType = upstream.headers.get("content-type") || "";
  
  const isM3U8 = explicitType === "m3u8" || target.split("?")[0].endsWith(".m3u8") || contentType.includes("mpegurl") || contentType.includes("m3u8");

  if (isM3U8 && request.method === "GET") {
    const text = await upstream.text();
    const finalUrl = upstream.url || target;
    const rewritten = rewriteManifest(text, finalUrl, customUa, customRef, customHeaders, url.origin);
    return new Response(rewritten, {
      status: upstream.status,
      headers: corsHeaders(allowedOriginHeader, { "content-type": "application/vnd.apple.mpegurl", "cache-control": "no-store" }),
    });
  }

  const passthroughHeaders = corsHeaders(allowedOriginHeader, {
    "content-type": contentType || "application/octet-stream",
    "cache-control": "public, max-age=15",
  });

  const contentRange = upstream.headers.get("content-range");
  if (contentRange) passthroughHeaders["content-range"] = contentRange;
  const acceptRanges = upstream.headers.get("accept-ranges");
  if (acceptRanges) passthroughHeaders["accept-ranges"] = acceptRanges;

  return new Response(upstream.body, { status: upstream.status, headers: passthroughHeaders });
}

function rewriteManifest(text, baseUrl, ua, ref, customHeaders, proxyOrigin = "") {
  const lines = text.split(/\r?\n/);
  let nextIsPlaylist = false;
  const out = lines.map((line) => {
    const trimmed = line.trim();
    if (!trimmed) return line;

    if (trimmed.startsWith("#")) {
      if (trimmed.startsWith("#EXT-X-STREAM-INF") || trimmed.startsWith("#EXT-X-I-FRAME-STREAM-INF")) {
        nextIsPlaylist = true;
      }
      if (trimmed.includes("URI=")) {
        return trimmed.replace(/URI="([^"]+)"/g, (_, p1) => {
          if (p1.startsWith('data:') || p1.startsWith('skd:') || p1.startsWith('blob:')) return `URI="${p1}"`;
          const abs = resolveUrl(p1, baseUrl);
          const params = new URLSearchParams();
          params.set("url", abs);
          if (ua) params.set("ua", ua);
          if (ref) params.set("ref", ref);
          if (customHeaders) params.set("headers", customHeaders);
          if (trimmed.startsWith("#EXT-X-MEDIA")) params.set("type", "m3u8");
          const encQ = xorBase64(params.toString(), SECRET_KEY);
          return `URI="${proxyOrigin}/proxy?q=${encodeURIComponent(encQ)}"`;
        });
      }
      return trimmed;
    }

    const abs = resolveUrl(trimmed, baseUrl);
    if (!nextIsPlaylist && (abs.includes('.mp4') || abs.includes('.ts') || abs.includes('.m4s') || abs.includes('.m4a') || abs.includes('brightcove'))) {
      return abs;
    }

    const params = new URLSearchParams();
    params.set("url", abs);
    if (ua) params.set("ua", ua);
    if (ref) params.set("ref", ref);
    if (customHeaders) params.set("headers", customHeaders);
    if (nextIsPlaylist) {
      params.set("type", "m3u8");
      nextIsPlaylist = false;
    }
    const encQ = xorBase64(params.toString(), SECRET_KEY);
    return `${proxyOrigin}/proxy?q=${encodeURIComponent(encQ)}`;
  });
  return out.join("\n");
}

function resolveUrl(u, base) {
  try { return new URL(u, base).toString(); } catch (e) { return u; }
}


function corsHeaders(origin = "*", extra = {}) {
  return {
    "Access-Control-Allow-Origin": origin,
    "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS, HEAD",
    "Access-Control-Allow-Headers": "Range, Content-Type, Authorization, Accept, User-Agent, Referer, Origin, X-Requested-With, Cache-Control",
    "Access-Control-Expose-Headers": "Content-Length, Content-Range, Content-Type, Accept-Ranges",
    ...extra,
  };
}

function json(obj, status = 200, origin = "*") {
  return new Response(JSON.stringify(obj), {
    status,
    headers: corsHeaders(origin, { "content-type": "application/json; charset=UTF-8" }),
  });
}

function xorBase64(text, key) {
  const encoder = new TextEncoder();
  const bytes = encoder.encode(text);
  const keyBytes = encoder.encode(key);
  for (let i = 0; i < bytes.length; i++) bytes[i] ^= keyBytes[i % keyBytes.length];
  let binary = "";
  for (let i = 0; i < bytes.length; i++) binary += String.fromCharCode(bytes[i]);
  return btoa(binary);
}

function dexorBase64(base64, key) {
  const binary = atob(base64);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
  const encoder = new TextEncoder();
  const keyBytes = encoder.encode(key);

  return new TextDecoder().decode(bytes);
}

async function loadPlaylist(url) {
  setLoadingState(true);
  els.list.innerHTML = '<div class="empty-state"><p class="empty-title">Memuat playlist...</p></div>';
  try {
    const res = await fetch('/api/playlist?url=' + encodeURIComponent(url));
    const data = await res.json();
    if (!res.ok || data.error) throw new Error(data.error || 'Gagal memuat playlist');
    
    let decodedData = data;
    if (data.encrypted) {
      try {
        decodedData = JSON.parse(dexorBase64(data.encrypted, SECRET_KEY));
      } catch(e) { throw new Error("Gagal dekripsi playlist"); }
    }
    allChannels = decodedData.channels || [];
    activeGroup = 'Semua';
    renderChips();
    renderList();
  } catch (err) {
    els.list.innerHTML = '<div class="empty-state"><p class="empty-title">Gagal memuat</p><p class="empty-body">' + escapeHtml(err.message) + '</p></div>';
    allChannels = [];
    els.chips.innerHTML = '';
    els.count.textContent = '';
  } finally {
    setLoadingState(false);
  }
}

        els.roStat.textContent = res + ' · ' + kbps + ' kbps';
      }
    });

    shakaPlayer.addEventListener('error', (event) => {
      if (event.detail && event.detail.fatal) {
        setStatus('error', 'DRM Error: ' + event.detail.code);
        console.error("Shaka Error Detail:", event.detail);
      }
    });

    shakaPlayer.load(ch.url).then(() => {
      updateShakaQualitySelect();
      els.video.play().catch(() => {});
      setStatus('live', hasDrm ? 'Live (DRM)' : 'Live (DASH)');
    }).catch(err => {
      setStatus('error', 'Gagal memuat DASH: ' + err.code);
    });

    els.video.addEventListener('playing', () => {
      if (shakaPlayer) setStatus('live', hasDrm ? 'Live (DRM)' : 'Live (DASH)');
    }, { once: false });

  } else {
    if (window.Hls && window.Hls.isSupported()) {
      hls = new Hls({ 
        enableWorker: true,
        maxBufferLength: 60,
        maxMaxBufferLength: 120,
        liveSyncDurationCount: 6, // Mundur 6 segmen (~30-60 detik) dari live edge
        liveMaxLatencyDurationCount: 15,
        fragLoadingMaxRetry: 10,
        manifestLoadingMaxRetry: 10,
        levelLoadingMaxRetry: 10,
        startLevel: -1 // Auto ABR dari awal
      });
      hls.loadSource(proxiedUrl);
      hls.attachMedia(els.video);

      hls.on(Hls.Events.MANIFEST_PARSED, () => {
        updateHlsQualitySelect();
        els.video.play().catch(() => {});
      });

      hls.on(Hls.Events.LEVEL_SWITCHED, (_, data) => {
        const level = hls.levels[data.level];
        if (level) {
          const res = level.width && level.height ? level.width + 'x' + level.height : '--x--';
          const kbps = level.bitrate ? Math.round(level.bitrate / 1000) : '--';
          els.roStat.textContent = res + ' · ' + kbps + ' kbps';
        }
      });

      hls.on(Hls.Events.ERROR, (_, data) => {
        if (data.fatal) setStatus('error', 'Error HLS: ' + data.type);
      });

      els.video.addEventListener('playing', () => {
        if (hls) setStatus('live', 'Live');
      }, { once: false });
    } else if (els.video.canPlayType('application/vnd.apple.mpegurl')) {
      // Fallback Native Player (biasanya Safari/iOS) - Quality Selector manual tidak didukung
      els.video.src = proxiedUrl;
      els.video.addEventListener('loadedmetadata', () => els.video.play().catch(() => {}), { once: true });
      els.video.addEventListener('playing', () => setStatus('live', 'Live'));
      els.video.addEventListener('error', () => setStatus('error', 'Error memutar'));
    } else {
      setStatus('error', 'HLS tidak didukung');
    }
  }
}

        mutations.forEach((mutation) => {
          if (mutation.type === 'attributes' && mutation.attributeName === 'style') {
            const target = mutation.target;
            if (target.style.display === 'none') {
              toggleChat(true); // Sembunyikan box utama
              target.style.display = 'block'; // Reset display Chatango agar bisa dibuka lagi
            }
          }
        });
      });

      const childObserver = new MutationObserver(() => {
        const chatangoDiv = chatContainer.querySelector('div');
        if (chatangoDiv) {
          styleObserver.observe(chatangoDiv, { attributes: true, attributeFilter: ['style'] });
          childObserver.disconnect();
        }
      });
      childObserver.observe(chatContainer, { childList: true });
    }
  }
})();

const server = http.createServer(async (req, res) => {
  try {
    const proto = req.headers['x-forwarded-proto'] || 'http';
    const url = new URL(req.url, `${proto}://${req.headers.host}`);
    let body = null;
    if (req.method !== 'GET' && req.method !== 'HEAD') {
      const chunks = [];
      for await (const chunk of req) chunks.push(chunk);
      body = Buffer.concat(chunks);
    }
    const headers = new Headers();
    for (let key in req.headers) {
      if (Array.isArray(req.headers[key])) {
        for (let val of req.headers[key]) headers.append(key, val);
      } else {
        headers.append(key, req.headers[key]);
      }
    }
    const webReq = new Request(url.href, {
      method: req.method,
      headers: headers,
      body: body,
      duplex: 'half'
    });
    
    const webRes = await handleRequest(webReq);
    
    const resHeaders = {};
    webRes.headers.forEach((v, k) => { resHeaders[k] = v; });
    res.writeHead(webRes.status, resHeaders);
    
    if (webRes.body) {
      for await (const chunk of webRes.body) {
        res.write(chunk);
      }
      res.end();
    } else {
      res.end();
    }
  } catch (e) {
    console.error(e);
    if (!res.headersSent) res.writeHead(500);
    res.end(e.toString());
  }
});
